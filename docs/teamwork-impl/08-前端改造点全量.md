## 二、★核心约束：**只能追加，不能改写**

这是本仓**已被验证**的手法（来自既有纪律）：

> **给"已被字符串锚定守卫固化的 UI 段"做视觉升级，首选「追加式覆盖」而非改写。**
> 在目标 CSS 段末尾追加一层规则，靠**同特异性后者胜**覆盖既有声明；不删除、不移动任何既有行
> ⇒ 所有以 `includes(...)` 断言源码文本的守卫**零破坏**。

**实测支撑**：本仓有约 **45 个 smoke 套件读取 `lib/client.js`**，其中大量以 `includes('data-dam-xxx')` 形式断言。

**因此前端团队化的正确做法**：

```js
// ★ 在 client.js 末尾（模块工厂内、return 之前）追加一整段，用醒目标记包裹：
/* ================= dam-team:begin ================= */
// 团队版 UI 全部在这里；不触碰上方任何既有行
/* ================= dam-team:end ================= */
```

**判据**：`dam-team:begin` 之前的**字节完全不变** —— 这条可以用 `git diff --stat` 直接验证（应显示纯新增行，零删除）。

## 三、逐项改动点

### 3.1 【必须】同步状态条

**位置**：追加段内，注册到既有布局槽位。
**数据源**：`POST /api/dsh-auto-memory/state` 的 `team` 字段（02 卷 `3.7 已设计）。

```js
/* ================= dam-team:begin ================= */
// —— 同步状态条：teamEnabled=false 或 team===null 时**不渲染任何东西**（零 UI 变化）——
function TeamStatusBar() {
  var tick = useTeamTick()                     // 见 `3.7 的轮询钩子
  var t = tick && tick.team
  if (!t) return null                          // ★ 关键：关闭时零渲染
  var pending = t.queue || 0
  var conflicts = t.conflicts || 0
  return h('div', { className: 'dam-team-bar', 'data-dam-team': 'bar' },
    h('span', { 'data-dam-team': 'member' }, t.member ? t.member.name : '未登录'),
    h('span', { 'data-dam-team': 'synced' }, '最后同步 ' + fmtAgo(t.syncAt)),
    pending > 0 ? h('span', { 'data-dam-team': 'queue' }, pending + ' 待同步') : null,
    conflicts > 0 ? h('span', {
      'data-dam-team': 'conflicts',           // ★ 有冲突时点击直达冲突中心
      style: { color: 'var(--dam-warn, #d97706)', cursor: 'pointer' },
      onClick: function () { openConflictCenter() },
    }, conflicts + ' 待裁决') : null
  )
}
/* ================= dam-team:end ================= */
```

**验收**：`teamEnabled=false` ⇒ `TeamStatusBar` 返回 `null` ⇒ **DOM 零变化**（既有 163 个锚点全在）。

### 3.2 【必须】记忆库加"作者/来源"标识

**位置**：既有记忆库列表渲染处 —— **不改既有渲染函数**，在追加段里做**后置装饰**：

```js
/* ================= dam-team:begin ================= */
// 后置装饰：给既有列表项补团队徽标（不碰既有渲染逻辑）
// 做法：监听既有列表容器的 DOM 变化，对带 data-dam-file 的行追加徽标
function decorateTeamBadges() {
  var items = document.querySelectorAll('[data-dam-file]')   // ★ 用既有锚点定位
  for (var i = 0; i < items.length; i++) {
    var el = items[i]
    if (el.querySelector('[data-dam-team-badge]')) continue  // 幂等
    var actor = el.getAttribute('data-dam-actor')            // 由后端在 list 响应里带上
    if (!actor) continue
    var b = document.createElement('span')
    b.setAttribute('data-dam-team-badge', actor)
    b.className = 'dam-team-badge'
    b.textContent = actor
    el.appendChild(b)
  }
}
// 观察 DOM 变化（只在团队启用时启动）
var damTeamObserver = null
function startBadgeObserver() {
  if (damTeamObserver || typeof MutationObserver === 'undefined') return
  damTeamObserver = new MutationObserver(function () { decorateTeamBadges() })
  damTeamObserver.observe(document.body, { childList: true, subtree: true })
}
/* ================= dam-team:end ================= */
```

**为什么用后置装饰而不是改渲染函数**：改渲染函数会碰约定被守卫断言的字符串；后置装饰**零侵入**。

⚠️ **风险**：`MutationObserver` 观察 `document.body` 在长列表下开销大 ⇒ **必须节流**（下面 `3.7 的 tick 里统一节流）。

### 3.3 【必须】冲突中心（4 类人工裁决）

**位置**：追加段，作为**浮层**（复用既有 DialogHost 的 kind 机制，或自建轻量浮层）。
**数据源**：`state.team.conflicts` 与详情接口。

```js
/* ================= dam-team:begin ================= */
// 冲突中心：只处理 4 类整篇级冲突（事实冲突已自动"保留双方"，不进这里）
var CONFLICT_KINDS = { rules: '用户规则', note: '项目笔记', plan: 'PLAN 白板', procedure: '技能晋升' }

function ConflictCenter(props) {
  var items = props.items || []
  if (!items.length) return h('div', { 'data-dam-team': 'conflict-empty' }, '暂无待裁决冲突')
  return h('div', { 'data-dam-team': 'conflict-center' },
    items.map(function (it) {
      return h('div', { key: it.id, 'data-dam-team': 'conflict-item', 'data-kind': it.kind },
        h('h4', null, CONFLICT_KINDS[it.kind] || it.kind),
        h('div', { className: 'dam-conflict-diff' },        // 双栏差异
          h('pre', { 'data-dam-team': 'mine' }, it.mine),
          h('pre', { 'data-dam-team': 'theirs' }, it.theirs)
        ),
        h('div', { className: 'dam-conflict-actions' },     // 三选一
          h('button', { 'data-dam-team': 'keep-mine',   onClick: function () { resolve(it.id, 'keep-mine') } },   '保留我的'),
          h('button', { 'data-dam-team': 'keep-theirs', onClick: function () { resolve(it.id, 'keep-theirs') } }, '采用对方'),
          h('button', { 'data-dam-team': 'keep-both',   onClick: function () { resolve(it.id, 'keep-both') } },   '都保留')
        )
      )
    })
  )
}
/* ================= dam-team:end ================= */
```

**回传**：`POST /api/dsh-auto-memory/handoff-permission`（复用既有路由）或新增 settle 字段。

### 3.4 【必须】团队设置分区（第 10 个分区）

**位置**：既有设置页 9 个分区的**追加**。做法：在追加段里把团队分区**追加进分区数组**（不修改既有 9 项）。

```js
/* ================= dam-team:begin ================= */
var TEAM_SETTING_KEYS = [
  'teamEnabled', 'teamServerUrl', 'teamId', 'teamProjectId', 'teamMemberName',
  'teamInjectEnabled', 'teamConflictPolicy', 'teamAuditEnabled', 'teamShowMemberBadges',
]
// 追加进既有分区列表（★ 用 push，不重建数组 —— 既有 9 项零改动）
function appendTeamSection(sections) {
  if (!sections || sections.some(function (s) { return s.key === 'team' })) return sections
  sections.push({
    key: 'team',
    title: '团队协作',
    'data-dam-team': 'settings-section',
    // ★ 即时回显：开关 onChange 立刻本地翻面，不等写盘回来（用户既有偏好）
    render: function () { return renderTeamSettings() },
  })
  return sections
}
/* ================= dam-team:end ================= */
```

**★ 即时回显判据（用户明确偏好）**：
> 「开关类改动必须即时回显（点击后标签立刻翻面），**写盘成功但界面无变化会被视为"功能坏了"**」

⇒ 开关必须**乐观更新**：点击 → 立刻改本地 state → 后台写盘 → 失败才回滚并提示。

### 3.5 【必须】成员在场指示

**数据源**：`state.team.member`（当前成员）；在场列表需团队服务端（可选）。

**降级判据**：**没有成员列表时只显示"我"**，不显示空列表 —— 空列表比不显示更糟。

### 3.7 【必须】数据轮询钩子（复用既有 tick 机制）

**关键约束**：**不得新增定时器**（会与既有 dispose 链冲突）。必须复用既有 tick。

```js
/* ================= dam-team:begin ================= */
// useTeamTick：复用既有 controller 的 tick（与既有 useTick 同源），只加一个节流
function useTeamTick() {
  // ★ 复用既有 useTick（client.js 内已有），而不是自己 setInterval
  var tick = useTick()                       // 既有钩子，零改动
  var team = (tick && tick.state && tick.state.team) || null
  return { team: team }
}
// MutationObserver 节流（避免长列表下每帧回调）
var _damBadgeTimer = null
function scheduleBadgeDecorate() {
  if (_damBadgeTimer) return
  _damBadgeTimer = setTimeout(function () { _damBadgeTimer = null; decorateTeamBadges() }, 120)
}
/* ================= dam-team:end ================= */
```

## 五、不可碰的清单（防误改）

| 类别 | 为什么 |
|---|---|
| **既有 13 个页签的渲染函数** | 被守卫断言源码文本 |
| **既有 163 个 `data-dam-*` 锚点** | 删除任何一个都会红 |
| **既有 23 个 `--dam-*` 变量** | 皮肤契约依赖 |
| **既有 9 个设置分区** | 同上 |
| **`__ModuleLoader__` 结构** | 改动会影响模块加载 |
| **ES5 语法** | 本仓前端不用 ES6+（不可引入 `const`/`let`/箭头函数） |

**★ 最后一条最容易被忽略**：追加段也必须用 **ES5 语法**（`var`、`function`），与既有代码一致。用 `const` 会让某些旧宿主构建路径炸。

## 六、前端验收（可脚本化）

```bash
node -e "const s=require('fs').readFileSync('lib/client.js','utf8');   const m=s.match(/\/\* =+ dam-team:begin =+ \*\/([\s\S]*?)\/\* =+ dam-team:end =+ \*\//);   if(!m) process.exit(0);   const hard=(m[1].match(/#[0-9a-fA-F]{3,6}/g)||[]).length;   console.log('追加段硬编码色:', hard); if(hard>0) process.exit(1)"

```
