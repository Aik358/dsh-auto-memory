# 04 · 同步原语与服务端契约（完整代码）

> 卷 03 给了 5 个新模块。本卷给**它们共同依赖的 6 个同步原语**，以及**服务端契约**。
> 6 个原语不是学术选择，而是**被插件既有数据结构逼出来的**：每条都能追到本仓的真实字段。

---

## 一、为什么是这 6 个：从真实数据结构反推

| 原语 | 名字 | 被哪个真实结构逼出来 | 出处 |
|---|---|---|---|
| **P1** | 字段级 LWW | 日历条目 `{date,time,quadrant,title,...}` | `index.js:8242 parseCalendar` / `:8280 calendarAdd` |
| **P2** | 追加日志 | 每日日志（append-only 明文） | 本仓硬纪律："绝不覆盖，只追加" |
| **P3** | OR-Set | 标签 / 技能阶段 / 成员集合 | `procedure-store` 的 stage 枚举 |
| **P4** | 两阶段移动 | 看板卡片跨列 / 白板区块跨泳道 | `board-mode.js` / `:8704 wbOwnerOf` |
| **P5** | 墓碑 | 条目删除（须留痕，不能真删） | 用户裁定："retracted 不是垃圾，是教训，不过滤只备注" |
| **P6** | 重算 | 向量 / 索引 / sidecar / 评分 | `wb-sidecar.js`（派生）/ `l0-index.js` |

## 二、六个原语的完整实现

### P1 —— 字段级 LWW（Last-Write-Wins）

**适用**：日历、成员资料、配置覆盖值 —— **字段独立、覆盖可接受**。

```js
// ===== 新增文件：lib/sync-primitives.js =====
/**
 * P1 字段级 LWW。
 *
 * ★ 关键设计：**不是整条 LWW，而是逐字段 LWW**。
 *   理由（真实场景）：A 改了日历的标题，B 改了同一条的时间 —— 整条 LWW 会丢掉一个改动；
 *   逐字段 LWW 两个都保住。本仓日历条目字段天然独立，适合逐字段。
 *
 * 混合时钟：{ wall: 毫秒时间戳, device: 设备 ID, seq: 设备内自增 }
 *   —— 单用 wall 会因设备时钟漂移出错；加 device 做确定性破平局。
 */
export function lwwMerge(mine, theirs, opts = {}) {
  if (!mine) return { value: theirs, changed: true, win: 'theirs' }
  if (!theirs) return { value: mine, changed: false, win: 'mine' }
  const out = { ...mine }
  const fields = opts.fields || Object.keys({ ...mine, ...theirs })
  let changed = false
  for (const f of fields) {
    if (f === '_clock') continue
    const a = mine[f], b = theirs[f]
    if (a === b) continue
    const ca = clockOf(mine, f), cb = clockOf(theirs, f)
    const bWins = compareClock(cb, ca) > 0
    if (bWins) { out[f] = b; changed = true }
  }
  // 合并时钟：逐字段取较新（用于下一轮比较）
  out._clock = mergeClocks(mine._clock, theirs._clock)
  return { value: out, changed, win: changed ? 'theirs' : 'mine' }
}

/** 比较混合时钟：wall 优先，wall 相同用 device 字典序破平局（确定性）。 */
function compareClock(a, b) {
  if (!a) return -1
  if (!b) return 1
  const aw = Number(a.wall) || 0, bw = Number(b.wall) || 0
  if (aw !== bw) return aw - bw
  const ad = String(a.device || ''), bd = String(b.device || '')
  if (ad !== bd) return ad < bd ? -1 : 1
  const as = Number(a.seq) || 0, bs = Number(b.seq) || 0
  return as - bs
}

function clockOf(obj, field) {
  const c = obj && obj._clock
  if (!c) return null
  return c[field] || c['*'] || null      // '*' = 整条级时钟（兜底）
}

function mergeClocks(a, b) {
  const out = { ...(a || {}) }
  for (const k of Object.keys(b || {})) {
    if (!out[k] || compareClock(b[k], out[k]) > 0) out[k] = b[k]
  }
  return out
}

/** 生成一次性时钟戳（写路径调用）。 */
export function stampClock(deviceId, seq) {
  return { wall: Date.now(), device: String(deviceId || 'dev'), seq: Number(seq) || 0 }
}
```

**回归风险**：本仓日历条目**当前没有 `_clock` 字段**。⇒ 新增字段必须**向后兼容**（缺字段时 `clockOf` 返回 null，`compareClock(null, x) = -1` ⇒ 视为最旧，不会误覆盖）。
**验收判据**：把 `_clock` 去掉后，`lwwMerge` 的行为退化为"总是 theirs 胜" ⇒ 必须有一条测试钉住这个退化行为是有意的。

### P2 —— 追加日志（Append-Only Log）

**适用**：每日日志、反思、证据链、审计流水 —— **只增不改，天然无冲突**。

```js
/**
 * P2 追加日志。
 *
 * ★ 本仓的日志/反思**本来就是 append-only 明文**（memory_log 的硬纪律）。
 *   ⇒ 团队同步不需要任何冲突解决：**把两边的行按 (时间, device, seq) 归并后重写即可**。
 *
 * 幂等键：每条日志行的 sha1(正文 + 日期 + 设备)。重复同步同一行不会产生重复。
 */
import { createHash } from 'node:crypto'

export function appendMerge(mineText, theirsText, opts = {}) {
  const myLines = splitEntries(mineText)
  const theirLines = splitEntries(theirsText)
  const seen = new Set(myLines.map((l) => l.key))
  const merged = myLines.slice()
  let added = 0
  for (const l of theirLines) {
    if (seen.has(l.key)) continue          // ★ 幂等去重
    merged.push(l); seen.add(l.key); added++
  }
  // 按 (sortKey, device) 稳定排序 —— 保证两端归并结果**逐字节一致**
  merged.sort((a, b) => (a.sortKey - b.sortKey) || (a.device < b.device ? -1 : a.device > b.device ? 1 : 0))
  return {
    text: merged.map((l) => l.raw).join('\n'),
    added, removed: 0,                      // 追加语义：永不删除
    changed: added > 0,
  }
}

/** 把日志切成"条目"：以二级标题/日期行为界；无界则整篇一条。 */
function splitEntries(text) {
  if (!text) return []
  const lines = String(text).split(/\r?\n/)
  const out = []
  let cur = []
  const flush = () => {
    if (!cur.length) return
    const raw = cur.join('\n')
    // 提取日期（YYYY-MM-DD）与设备标记（如有）
    const dateM = raw.match(/\b(\d{4}-\d{2}-\d{2})\b/)
    const devM = raw.match(/<!--\s*dam-device:([^>]+?)\s*-->/)
    out.push({
      raw, key: sha1(raw),                  // ★ 内容寻址：同内容必同键
      sortKey: dateM ? Date.parse(dateM[1]) || 0 : 0,
      device: devM ? devM[1].trim() : '',
    })
    cur = []
  }
  for (const ln of lines) {
    if (/^##\s/.test(ln) && cur.length) flush()
    cur.push(ln)
  }
  flush()
  return out
}

function sha1(s) { return createHash('sha1').update(s, 'utf8').digest('hex').slice(0, 16) }
```

**判据**：**"两端归并结果逐字节一致"** —— 由 `merged.sort` 的稳定排序保证。必须有测试：把同样的两段文本以**相反顺序**喂进去，输出必须相同。

### P3 —— OR-Set（可加可减集合）

**适用**：标签、技能阶段、成员角色集合 —— **多值、需支持"删掉我加的那个"**。

```js
/**
 * P3 OR-Set（Observed-Remove Set）。
 *
 * 为什么不用普通 Set：普通集合**无法区分"我没看到过"和"我被删了"**。
 *   A 加标签 X，B 同时删标签 X ⇒ 普通集合会互相覆盖，结果不定。
 *   OR-Set 给每次"添加"发一个唯一 tag；删除 = 撤回我**已观察到**的 tag。
 *   ⇒ 未被观察到的添加不会被误删（这正是我们要的语义）。
 *
 * 数据结构：{ add: { tag: value }, remove: Set<tag> }
 */
export function orSetMerge(mine, theirs) {
  const a = norm(mine), b = norm(theirs)
  const add = { ...a.add }
  for (const [tag, val] of Object.entries(b.add)) {
    if (!(tag in add)) add[tag] = val
  }
  // remove 取并集（撤回是单调的：一旦撤回，永远撤回）
  const remove = new Set([...a.remove, ...b.remove])
  return { add, remove, changed: JSON.stringify(add) !== JSON.stringify(a.add) || remove.size !== a.remove.size }
}

/** 当前可见值 = add 中排除已被 remove 的 tag。 */
export function orSetValues(s) {
  const n = norm(s)
  const out = new Set()
  for (const [tag, val] of Object.entries(n.add)) {
    if (!n.remove.has(tag)) out.add(val)
  }
  return [...out]
}

/** 添加：生成唯一 tag（设备 + 序号 + 时间 ⇒ 全局唯一且可排序）。 */
export function orSetAdd(s, val, deviceId) {
  const n = norm(s)
  const tag = String(deviceId || 'dev') + ':' + Date.now().toString(36) + ':' + (Object.keys(n.add).length + 1)
  n.add[tag] = val
  return n
}

/** 删除：只撤回**我观察到的**那些 tag（未观察到的保留 ⇒ 不误删他人的新增）。 */
export function orSetRemove(s, val, observed) {
  const n = norm(s)
  const tags = (observed || Object.keys(n.add)).filter((t) => n.add[t] === val)
  for (const t of tags) n.remove.add(t)
  return n
}

function norm(s) {
  return {
    add: (s && s.add) || {},
    remove: new Set((s && s.remove) ? [...s.remove] : []),
  }
}

/** 序列化用（Set 不能直接进 JSON）。 */
export function orSetToJson(s) { const n = norm(s); return { add: n.add, remove: [...n.remove] } }
export function orSetFromJson(j) { return norm(j) }
```

### P4 —— 两阶段移动（Two-Phase Move）

**适用**：看板卡片跨列、白板区块跨泳道、技能库晋升 —— **涉及"所有权移交"**。

**为什么必须两阶段**：本仓的看板/白板有**归属门（`wbOwnerOf`）**。直接移动会撞门：

```js
// 本仓既有实现（lib/index.js:8692 附近）—— 归属判据
wbOwnerOf(sid, st)   // → 'current' | 'sealed' | 'orphan'
// 期牌比对用 _wbTokenOf 只读落盘值、不做派生兜底
```

⇒ 跨机移动必须：**① 先在源端"封印"（seal）→ ② 在目标端"认领"（claim）**。中间态可见且可恢复。

```js
/**
 * P4 两阶段移动。
 *
 * 阶段① propose：源端标记"待移交"，**保留原位**（不立即删）—— 失败可回滚。
 * 阶段② commit ：目标端确认接收后，源端才真正移除；同时写墓碑。
 *
 * 幂等：同一 moveId 重复 propose/commit 均无副作用。
 * 超时：propose 后 N 分钟未 commit ⇒ 自动回滚（源端解除封印）。
 */
export function createMover({ outbox, diag, sealOwner, claimOwner, now = () => Date.now() }) {
  const pending = new Map()          // moveId → { from, to, itemId, at, sealed }
  const TTL_MS = 10 * 60 * 1000

  return {
    /**
     * 阶段①：发起移动。
     * @returns {{ok:boolean, moveId?:string, reason?:string}}
     */
    propose({ itemId, from, to, actor }) {
      const moveId = 'mv_' + now().toString(36) + '_' + Math.random().toString(36).slice(2, 8)
      // ★ 先封印源端（走既有归属门，不绕过）
      let sealed = false
      try { sealed = sealOwner ? sealOwner(from, itemId, moveId) !== false : true } catch (e) {
        try { diag('move seal failed: ' + ((e && e.message) || e)) } catch (_) {}
        return { ok: false, reason: 'seal-failed' }
      }
      pending.set(moveId, { from, to, itemId, at: now(), sealed, actor })
      // 出队：让对端知道有个待认领的移动
      outbox.enqueue({ op: 'move-propose', target: itemId, actor, kind: 'move', moveId, from, to, at: now() })
      return { ok: true, moveId }
    },

    /** 阶段②：目标端确认。 */
    commit(moveId) {
      const p = pending.get(moveId)
      if (!p) return { ok: false, reason: 'unknown-move' }
      if (now() - p.at > TTL_MS) { this.rollback(moveId); return { ok: false, reason: 'expired' } }
      let claimed = false
      try { claimed = claimOwner ? claimOwner(p.to, p.itemId, moveId) !== false : true } catch (e) {
        try { diag('move claim failed: ' + ((e && e.message) || e)) } catch (_) {}
        return { ok: false, reason: 'claim-failed' }
      }
      if (!claimed) return { ok: false, reason: 'claim-refused' }
      pending.delete(moveId)
      outbox.enqueue({ op: 'move-commit', target: p.itemId, kind: 'move', moveId, at: now() })
      return { ok: true }
    },

    /** 回滚：解除源端封印（幂等）。 */
    rollback(moveId) {
      const p = pending.get(moveId)
      if (!p) return { ok: true, noop: true }
      pending.delete(moveId)
      try { if (p.sealed && sealOwner && sealOwner.unseal) sealOwner.unseal(p.from, p.itemId, moveId) } catch (_) {}
      outbox.enqueue({ op: 'move-rollback', target: p.itemId, kind: 'move', moveId, at: now() })
      return { ok: true }
    },

    /** 超时巡检（挂在既有的心跳巡检上，不新增定时器）。 */
    sweep() {
      const dead = []
      for (const [id, p] of pending) if (now() - p.at > TTL_MS) dead.push(id)
      for (const id of dead) this.rollback(id)
      return { rolledBack: dead.length }
    },

    state() { return { pending: pending.size, items: [...pending.keys()] } },
  }
}
```

**★ 与既有归属门的关系**：`sealOwner` / `claimOwner` 由 `index.js` 注入，内部走**既有的 `wbOwnerOf` / `_wbTokenOf`**，**不新写一套归属判据**（否则会重犯"同一语义多处各写一份"的已知缺陷）。

### P5 —— 墓碑（Tombstone）

**适用**：条目删除、技能下架、规则条目移除 —— **删了但必须留痕**。

```js
/**
 * P5 墓碑。
 *
 * ★ 用户的既有裁定直接决定了本原语：
 *   「superseded 的可见后果 = 返回但标记（不剔除、不降权）」
 *   「retracted 不是垃圾，是教训，不过滤只备注」
 *   ⇒ 团队同步里**永远不做物理删除**，只写墓碑；检索到墓碑条目时照常返回但标记。
 *
 * 墓碑形态：{ id, deletedAt, by, reason, kind: 'retracted'|'superseded'|'deleted' }
 * 合并：墓碑**单调**（两个墓碑取并集），无冲突可能。
 */
export function tombstoneMerge(mine, theirs) {
  const out = new Map()
  for (const t of [...(mine || []), ...(theirs || [])]) {
    if (!t || !t.id) continue
    const prev = out.get(t.id)
    if (!prev || (Number(t.deletedAt) || 0) > (Number(prev.deletedAt) || 0)) out.set(t.id, t)
  }
  return [...out.values()]
}

/** 删除 = 写墓碑（绝不 fs.rm / 绝不 splice 掉记录）。 */
export function makeTombstone({ id, by, kind = 'deleted', reason = '' }) {
  return { id: String(id), deletedAt: Date.now(), by: String(by || ''), kind, reason: String(reason || '') }
}

/** 查询时应用墓碑：返回但标记（**不剔除** —— 这是用户裁定，不是妥协）。 */
export function applyTombstones(items, tombstones) {
  const byId = new Map((tombstones || []).map((t) => [String(t.id), t]))
  return (items || []).map((it) => {
    const t = byId.get(String(it && it.id))
    if (!t) return it
    return {
      ...it,
      _tombstone: t,
      // 给前端/模型的显式标记（可直接渲染成"已撤回"徽标）
      _statusLabel: t.kind === 'retracted' ? '已撤回' : t.kind === 'superseded' ? '已被取代' : '已删除',
      _statusNote: t.reason || '',
    }
  })
}
```

### P6 —— 重算（Recompute）

**适用**：向量、索引、sidecar、评分、统计 —— **绝不传输，永远本地重算**。

```js
/**
 * P6 重算。
 *
 * ★ 这是**缩小同步面的第一刀**：以下数据永不进同步通道，收到对端变更后本地重算。
 *   依据（逐条可追）：
 *   · 向量/嵌入   —— 大（数 MB），且模型版本不同则不可比 ⇒ 必须本机重算
 *   · l0 索引     —— 纯派生，见 l0-index.js / l0-index-sync.js
 *   · wb-sidecar  —— 已明确是派生结构（wb-sidecar.js）
 *   · recall 评分 —— recall-stats.js 统计量，重算即可
 *   · 工作区概览  —— ws-overview-rank.js 排序
 */
export const DERIVED_KINDS = Object.freeze([
  'vector', 'embedding', 'l0-index', 'index', 'sidecar', 'score', 'stats', 'rank', 'overview',
])

export function isDerived(kind) { return DERIVED_KINDS.includes(String(kind)) }

/**
 * 重算调度器。判据：**触发式重算 + 防抖 + 单飞**，绝不在注入关键路径上同步重算。
 */
export function createRecomputeScheduler({ recompute, diag, debounceMs = 2000 }) {
  const dirty = new Set()
  let timer = null
  let running = false

  async function run() {
    if (running) return
    running = true
    try {
      const kinds = [...dirty]
      dirty.clear()
      for (const k of kinds) {
        try { await recompute(k) } catch (e) {
          try { diag('recompute ' + k + ' failed: ' + ((e && e.message) || e)) } catch (_) {}
          dirty.add(k)                       // 失败重新标脏，下次再试
        }
      }
    } finally { running = false }
  }

  return {
    /** 标记某类派生数据变脏（幂等）。 */
    mark(kind) {
      if (!isDerived(kind)) return { ok: false, reason: 'not-derived' }
      dirty.add(kind)
      if (timer) clearTimeout(timer)
      timer = setTimeout(() => { timer = null; void run() }, debounceMs)
      return { ok: true, pending: dirty.size }
    },
    flush() { if (timer) { clearTimeout(timer); timer = null } return run() },
    state() { return { dirty: [...dirty], running } },
  }
}
```

## 三、服务端契约（REST + CDC）

**设计约束（来自 01 卷的实测结论）**：
- 插件**不能接收服务端入站连接**（56 条路由全被回环守卫锁死）⇒ **只能客户端拉**。
- ⇒ 协议必须是 **"客户端推 + 客户端拉游标"**（CDC = Change Data Capture），**不是** WebSocket 推送。

### 3.1 端点表（4 个，全部由客户端发起）

| 方法 | 路径 | 用途 | 幂等 |
|---|---|---|---|
| `POST` | `/v1/apply` | 上行：批量提交本机变更 | ✅ 靠 `idemKey` |
| `GET` | `/v1/changes?since=<cursor>&team=<id>` | 下行：拉增量 | ✅ 只读 |
| `GET` | `/v1/members` | 成员与角色 | ✅ 只读 |
| `POST` | `/v1/conflicts/resolve` | 人工裁决回传 | ✅ 靠 `conflictId` |

**每个请求必带**（3 个头，缺一不可）：
```
Authorization: Bearer <宿主提供的 token>      ← 身份（团队身份来自 team-identity.js）
x-dam-team: <teamId>                          ← 租户隔离（服务端 where 条件，客户端不过滤）
x-dam-client: dsh-auto-memory/3.1.7           ← 版本（服务端可据此做兼容与灰度）
```

## A.2 两个真 bug（根因 + 修法）

### BUG-1（P1）：`compareClock` 在**两边都无时钟**时返回 `-1`，导致两端**分叉**

```js
// ❌ 有缺陷的写法
function compareClock(a, b) {
  if (!a) return -1        // ← 问题在这一行
  if (!b) return 1
  ...
}
```

**推演出的后果**（不只是"对方值不生效"，更严重的是**破坏收敛**）：

| 端 | 本站 | 收到 | `compareClock(对方, 本站)` | 结论 |
|---|---|---|---|---|
| A | `{title:'old'}` | `{title:'new'}` | `compareClock(null, null) = -1` | 认为**对方旧** ⇒ 保留 old |
| B | `{title:'new'}` | `{title:'old'}` | `compareClock(null, null) = -1` | 认为**对方旧** ⇒ 保留 new |

⇒ **A 停在 old、B 停在 new，两端永久分叉。** 这是同步算法里最不能犯的错（比丢数据更隐蔽：两端都"觉得自己对"）。

**修法**：两边都无时钟 ⇒ 返回 **0**（平局），再用**内容哈希**做确定性破平局，保证两端算出同一个答案。

### BUG-2（P2）：排序键不足，**同键同设备时依赖输入顺序** ⇒ 两端结果不一致

```js
// ❌ 有缺陷的写法：只有两级排序
merged.sort((a, b) => (a.sortKey - b.sortKey) || (a.device < b.device ? -1 : a.device > b.device ? 1 : 0))
```

**后果**：当日志行**没有日期**（`sortKey=0`）且**没有设备标记**（`device=''`）时，两级键全部相等 ⇒ `Array.sort` 保持**输入顺序** ⇒ **A→B 与 B→A 得到不同结果**，两端文本不同 ⇒ **每次同步都产生新差异，永远收敛不了**。

**修法**：加**第三级排序键 = 行内容哈希**（内容寻址，跨机一致）。

### 附带的期望值错误（我的测试写错了）

第 4 条我期望"3 条"，实际 **4 条才是对的**：A 有 2 条 bullet（`- A 的条目` + `- A 第二天`），B 有 2 条（`- B 的条目` + `- B 第三天`），合起来 4 条。**是我的期望值写错，不是代码错。**

## A.3 修好的实现（可直接用）

```js
import { createHash } from 'node:crypto'
const sha = (s) => createHash('sha1').update(s, 'utf8').digest('hex').slice(0, 16)

// ============ P1 字段级 LWW（已修 BUG-1） ============
function compareClock(a, b) {
  if (!a && !b) return 0            // ★ FIX-1：都无时钟 = 平局（原为 -1 ⇒ 两端分叉）
  if (!a) return -1
  if (!b) return 1
  const aw = Number(a.wall) || 0, bw = Number(b.wall) || 0
  if (aw !== bw) return aw - bw
  const ad = String(a.device || ''), bd = String(b.device || '')
  if (ad !== bd) return ad < bd ? -1 : 1
  return (Number(a.seq) || 0) - (Number(b.seq) || 0)
}
function clockOf(o, f) { const c = o && o._clock; return c ? (c[f] || c['*'] || null) : null }
function mergeClocks(a, b) {
  const o = { ...(a || {}) }
  for (const k of Object.keys(b || {})) if (!o[k] || compareClock(b[k], o[k]) > 0) o[k] = b[k]
  return o
}

export function lwwMerge(mine, theirs, opts = {}) {
  if (!mine) return { value: theirs, changed: true, win: 'theirs' }
  if (!theirs) return { value: mine, changed: false, win: 'mine' }
  const o = { ...mine }
  let changed = false
  const fields = opts.fields || Object.keys({ ...mine, ...theirs })
  for (const f of fields) {
    if (f === '_clock') continue
    const mv = mine[f], tv = theirs[f]
    if (mv === tv) continue
    let theirsWins
    const cc = compareClock(clockOf(theirs, f), clockOf(mine, f))
    if (cc !== 0) theirsWins = cc > 0
    else {
      // ★ FIX-2：无时钟或时钟相等 ⇒ 内容哈希确定性破平局（两端算出同一答案）
      theirsWins = sha(JSON.stringify(tv)) > sha(JSON.stringify(mv))
    }
    if (theirsWins) { o[f] = tv; changed = true }
  }
  o._clock = mergeClocks(mine._clock, theirs._clock)
  return { value: o, changed, win: changed ? 'theirs' : 'mine' }
}
```

```js
function splitEntries(text) {
  if (!text) return []
  const lines = String(text).split(/\r?\n/)
  const acc = []; let cur = []
  const flush = () => {
    if (!cur.length) return
    const raw = cur.join('\n')
    const dm = raw.match(/\b(\d{4}-\d{2}-\d{2})\b/)
    const dv = raw.match(/<!--\s*dam-device:([^>]+?)\s*-->/)
    acc.push({
      raw, key: sha(raw),                                       // ★ 内容寻址键
      sortKey: dm ? Date.parse(dm[1]) || 0 : 0,
      device: dv ? dv[1].trim() : '',
    })
    cur = []
  }
  for (const ln of lines) { if (/^##\s/.test(ln) && cur.length) flush(); cur.push(ln) }
  flush()
  return acc
}

export function appendMerge(mineText, theirsText) {
  const myL = splitEntries(mineText), thL = splitEntries(theirsText)
  const seen = new Set(myL.map((l) => l.key))
  const merged = myL.slice()
  let added = 0
  for (const l of thL) { if (seen.has(l.key)) continue; merged.push(l); seen.add(l.key); added++ }
  merged.sort((a, b) =>
      (a.sortKey - b.sortKey)
   || (a.device < b.device ? -1 : a.device > b.device ? 1 : 0)
   || (a.key < b.key ? -1 : a.key > b.key ? 1 : 0))   // ★ FIX-3：第三级 = 内容哈希
  return { text: merged.map((l) => l.raw).join('\n'), added, entries: merged.length }
}
```

## A.5 这两条 bug 教给实施者的三件事（请写进代码注释）

1. **同步算法的验收判据不是"能不能合并"，而是"两端是否收敛到同一结果"。**
   ⇒ 每条 merge 函数都**必须有一个"交换输入顺序 ⇒ 输出逐字节一致"的测试**。我这两条 bug 都是**只有做交换测试才能发现**的 —— 单向测试会全绿。

2. **`compareClock` 这类"比较器"的 null 语义是最容易埋雷的地方。**
   `if (!a) return -1` 看起来是常识，实际在"双方都缺"时会**让双方都认为对方旧**。
   ⇒ 比较器必须**显式处理 (a 空, b 空) 这个组合**，不能靠 `if` 链的顺序侥幸。

3. **`Array.prototype.sort` 在比较器返回 0 时保留输入顺序 ⇒ 输入顺序成了隐式输入。**
   ⇒ 任何要跨机收敛的排序，**比较器必须给出全序**（键用尽时用内容哈希兜底）。
