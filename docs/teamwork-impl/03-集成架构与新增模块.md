## 一、先说三个"已存在"的东西 —— 不要重复造

读源码的最大收获：**插件已经为团队化埋好了三块地基**，Teamwork 只需要接上去。

### 1.1 `fact-store.js` 已有冲突原语（★最重要发现）

```js
// lib/fact-store.js:127-137  原文照抄
/**
 * 冲突定义: 同一 subject+predicate+scope 但 object 不同(或一方有 object 一方无)。
 * 注意: 完全相同的 (subject,predicate,object,scope) 不算冲突, 是重复(走 merge)。
 */
export function isFactConflict(a, b) {
  if (!a || !b) return false
  if (a.scope !== b.scope || a.subject !== b.subject || a.predicate !== b.predicate) return false
  const ao = a.object === undefined || a.object === null ? '' : a.object
  const bo = b.object === undefined || b.object === null ? '' : b.object
  return ao !== bo
}
```

**这意味着**：事实库**天生就是"主谓宾三元组"**结构（`subject + predicate + object + scope`），而不是自由文本。
⇒ **团队同步的冲突检测不需要新发明算法**，直接复用这个三元组键即可：

```
冲突键（同步用） = sha1(scope + '\u0000' + subject + '\u0000' + predicate)
冲突内容        = object
```

**同一键、不同 object = 冲突**；**同键同 object = 重复（merge）**。这是**内容寻址**，天然幂等、天然去重、天然跨机一致。

### 2.1 `lib/team-identity.js` —— 成员身份（不建账号体系）

```js
/**
 * 团队身份：把**宿主提供的登录态**映射成团队成员（actor）。
 *
 * 设计铁律（来自用户裁定 + 宿主实测）：
 *  ① **不自建账号体系** —— 登录由宿主 ctx.authorization / dsh-deepseek-account 提供；
 *  ② 插件只做三件事：**映射成员 / 注入 actor / 记录审计**；
 *  ③ 宿主未登录时，插件必须**仍可单人使用**（团队功能整体置灰，不影响本机）。
 *
 * 依赖（全部为宿主 seam，读不到就降级，绝不抛）：
 *  · ctx.deepseekAccount —— 账号状态（signed-out 事件）
 *  · ctx.authorization    —— 凭据接缝（团队 token 若需要）
 *  · ctx.credentials      —— 凭据存取（插件不落明文）
 */
export function createTeamIdentity({ ctx, engine, diag }) {
  let cached = null            // { id, name, role, teamId, at }
  let lastErr = null

  /** 读宿主账号 → 映射成员。全程 try/catch，返回 null 表示"当前无身份"。 */
  function currentMember() {
    try {
      // ① 优先读宿主账号（真源）
      const acct = ctx && ctx.deepseekAccount
      const stateOf = acct && typeof acct.state === 'function' ? acct.state() : null
      const signedIn = !!(stateOf && (stateOf.signedIn || stateOf.user))
      const user = (stateOf && stateOf.user) || null
      if (!signedIn || !user) return null

      // ② 映射为团队成员（id 用宿主侧稳定标识，**不用显示名** —— 显示名可改）
      const stableId = String(user.id || user.sub || user.email || '').trim()
      if (!stableId) return null
      const name = String(user.name || user.email || stableId).trim()

      // ③ 角色来自团队服务端下发（本机不猜）；未同步到时给最小权限
      const role = (cached && cached.teamId === engine.config.teamId && cached.id === stableId)
        ? cached.role : 'member'

      cached = { id: stableId, name, role, teamId: engine.config.teamId || '', at: Date.now() }
      return cached
    } catch (e) {
      lastErr = String((e && e.message) || e)
      try { diag('team identity read failed: ' + lastErr) } catch (_) {}
      return null
    }
  }

  /** 供工具信封调用：给每次写入打上 actor。 */
  function currentActor(agent) {
    const m = currentMember()
    if (!m) return null
    return {
      id: m.id,
      name: m.name,
      role: m.role,
      // 设备与工作区：用于「同一个人两台机器」的区分
      device: engine.deviceIdPre ? engine.deviceIdPre() : '',
      ws: (agent && agent.session && agent.session.header && agent.session.header.cwd) || '',
    }
  }

  return {
    currentMember,
    currentActor,
    /** 退出登录时清缓存（订阅宿主事件；失败不影响）。 */
    install() {
      try {
        if (ctx && typeof ctx.on === 'function') {
          ctx.on('deepseek-account/signed-out', () => { cached = null })
        }
      } catch (_) {}
    },
    status() { return { member: currentMember(), error: lastErr } },
  }
}
```

### 2.2 `lib/team-outbox.js` —— 本机变更队列（离线可写）

```js
import fs from 'node:fs'
import path from 'node:path'
import { writeTextAtomicPreSync } from './config-io.js'

/**
 * 团队出站队列（outbox）。核心不变量（逐条都要有测试）：
 *  ① **写本机永不被队列阻塞** —— enqueue 是同步内存操作 + 异步落盘；
 *  ② **幂等** —— 幂等键 = {op, targetKey, actorId, lamport}，重复 enqueue 自动去重；
 *  ③ **重启可续** —— 每次变更后同步落盘（原子写，带 pid+序号，遵既有纪律）；
 *  ④ **有界** —— 超过 maxItems 时按"最旧先丢 + 记降级"处理，绝不无限增长。
 */
export function createTeamOutbox({ engine, dir, maxItems = 5000, diag }) {
  const file = path.join(dir, 'team-outbox.json')
  let items = []
  let lamport = 0
  let seq = 0
  let loaded = false

  function load() {
    if (loaded) return
    loaded = true
    try {
      const raw = fs.readFileSync(file, 'utf8')
      const j = JSON.parse(raw)
      items = Array.isArray(j.items) ? j.items : []
      lamport = Number(j.lamport) || 0
      seq = Number(j.seq) || 0
    } catch (_) { items = [] }
  }

  function persistSync() {
    try {
      const body = JSON.stringify({ v: 1, lamport, seq, items })
      // ★ 同步原子写：dispose 阶段不保证异步完成，必须同步
      writeTextAtomicPreSync(file, body)
    } catch (e) { try { diag('outbox persist failed: ' + ((e && e.message) || e)) } catch (_) {} }
  }

  /** 幂等键：同一 (op,target) 在同一逻辑时刻只入队一次。 */
  function idemKey(it) { return it.op + '|' + it.target + '|' + it.actor + '|' + it.at }

  return {
    load,
    persistSync,

    /** 入队。同步返回，绝不抛。 */
    enqueue(it) {
      load()
      if (!it || !it.op || !it.target) return { ok: false, reason: 'malformed' }
      const key = it.kind + '|' + idemKey(it)
      if (items.some((x) => x._k === key)) return { ok: true, deduped: true }
      lamport += 1
      const rec = { ...it, lamport, _k: key, seq: ++seq }
      items.push(rec)
      if (items.length > maxItems) {
        const dropped = items.splice(0, items.length - maxItems)
        try { diag('outbox overflow, dropped ' + dropped.length) } catch (_) {}
      }
      persistSync()                 // ★ 立即落盘：崩溃可续
      return { ok: true, lamport, pending: items.length }
    },

    /** 取出待推送（不删除；推送成功后再 ack）。 */
    peek(limit = 100) { load(); return items.slice(0, limit) },

    /** 推送成功后确认移除。 */
    ack(keys) {
      load()
      const set = new Set(keys || [])
      const before = items.length
      items = items.filter((x) => !set.has(x._k))
      if (items.length !== before) persistSync()
      return { ok: true, removed: before - items.length, pending: items.length }
    },

    size() { load(); return items.length },
    snapshot() { load(); return { pending: items.length, lamport, seq } },
  }
}
```

### 2.3 `lib/team-pull.js` —— 下行合并（只走既有写路径）

```js
/**
 * 团队拉取与合并。**最重要的设计判据**：
 *  下行数据**必须走既有的、带校验的写路径**，不得绕过 Gate 直接写文件。
 *  理由：本仓的记忆卫生门（hygieneGateForPrimitive，:11257）、事实性判据、
 *        MutationBoundary 校验都是为了防「脏记忆入册」；绕过它们等于把防线拆了。
 */
export function createTeamPuller({ engine, fetchJson, conflictCenter, outbox, diag }) {
  let since = 0                    // 服务端游标
  let inflight = false

  return {
    /** 拉一次增量。返回 {ok, changed, applied, rejected}；永不抛。 */
    async pullOnce() {
      if (inflight) return { ok: false, reason: 'inflight' }
      if (!engine.config.teamEnabled || !engine.config.teamServerUrl) return { ok: false, reason: 'disabled' }
      inflight = true
      try {
        const r = await fetchJson('/v1/changes?since=' + since)
        if (!r.ok) return { ok: false, reason: r.error }
        const changes = (r.data && r.data.changes) || []
        let applied = 0, rejected = 0
        for (const ch of changes) {
          try {
            if (this.applyOne(ch)) applied++
            else rejected++
          } catch (e) {
            rejected++
            try { diag('pull apply failed: ' + ((e && e.message) || e)) } catch (_) {}
          }
        }
        since = (r.data && r.data.cursor) || since
        if (rejected) conflictCenter.recordRejected(rejected)   // 拒绝也留痕
        return { ok: true, changed: applied > 0, applied, rejected }
      } catch (e) {
        return { ok: false, reason: String((e && e.message) || e) }
      } finally { inflight = false }
    },

    /**
     * 应用单条变更。**判据：能走既有引擎方法的，一律走既有方法。**
     * 返回 true = 已应用；false = 被既有校验拒绝（正常业务结果，不是错误）。
     */
    applyOne(ch) {
      if (!ch || !ch.kind) return false
      switch (ch.kind) {
        case 'fact': {
          // ★ 走既有 fact-store 的 upsert —— 冲突判定/合并/六态结果全部复用
          const res = engine._factStore && engine._factStore.upsert
            ? engine._factStore.upsert(ch.payload)     // 既有签名
            : null
          if (!res) return false
          if (res.outcome === 'conflict-added') conflictCenter.push(res)   // 冲突入队待人工
          return res.ok !== false
        }
        case 'procedure': {
          // ★ 走既有 scoped 双库分派
          const scoped = engine._scopedProcedureIo
          if (!scoped || typeof scoped.write !== 'function') return false
          scoped.write(ch.payload)                       // 既有 API 自己管物理分文件
          return true
        }
        case 'handoff': {
          // 账本是**追加新篇**：不覆盖，直接落文件（P2 语义）
          return engine._appendHandoffPre ? (engine._appendHandoffPre(ch.payload), true) : false
        }
        default:
          return false                                   // 未知类型：不猜、不写、留痕
      }
    },

    state() { return { since, inflight } },
  }
}
```

### 2.4 `lib/team-conflict.js` —— 冲突中心（只对 4 处人工裁决）

```js
/**
 * 冲突中心。设计判据：**只对"整篇级/治理级"的 4 处建队列**，
 * 其余冲突（事实三元组、证据追加、日志）全部可自动解，不要塞进来刷屏。
 *
 * 4 处需人工裁决：
 *   ① 用户规则整篇（MEMORY.md 层级改动）
 *   ② 项目笔记整篇（MEMORY.md 被两人同时重写）
 *   ③ PLAN 白板整篇重写
 *   ④ 通用库技能审批（晋升）
 */
export function createConflictCenter({ engine, dir, diag }) {
  const items = []          // {id, kind, mine, theirs, at, resolved}

  return {
    /** 事实三元组冲突：**自动保留双方**（本仓既有 semantics），只记账不阻塞。 */
    push(res) {
      try {
        items.push({
          id: 'c_' + Date.now().toString(36) + '_' + (items.length + 1),
          kind: 'fact',
          autoResolved: 'both-kept',      // ★ 事实冲突默认「保留双方 provenance」
          payload: res,
          at: Date.now(),
        })
      } catch (_) {}
    },

    /** 整篇级冲突：需要人工裁决（白板/笔记/规则）。 */
    pushManual(kind, mine, theirs) {
      const it = {
        id: 'm_' + Date.now().toString(36) + '_' + (items.length + 1),
        kind, autoResolved: null, mine, theirs, at: Date.now(), resolved: null,
      }
      items.push(it)
      try { engine._notifyConflictPre && engine._notifyConflictPre(it) } catch (_) {}
      return it
    },

    /** 拒绝留痕：下行被既有校验拒掉的数据也要可见（否则"静默丢数据"）。 */
    recordRejected(n) {
      try { diag('team pull rejected ' + n + ' change(s)') } catch (_) {}
    },

    scan(_changed) { /* 预留：批量变更后的二轮一致性检查 */ },

    list() { return items.slice(-200) },
    pendingManual() { return items.filter((x) => !x.autoResolved && !x.resolved) },

    /** 人工裁决：keep-mine | keep-theirs | keep-both | per-block（白板逐区块）。 */
    resolve(id, action) {
      const it = items.find((x) => x.id === id)
      if (!it) return { ok: false, reason: 'not-found' }
      it.resolved = { action, at: Date.now(), by: engine._actorNamePre ? engine._actorNamePre() : '' }
      return { ok: true, item: it }
    },
  }
}
```

### 2.5 `lib/team-project-map.js` —— 团队项目 ID（解决 wsKey 跨机不汇合）

```js
import fs from 'node:fs'
import path from 'node:path'
import { createHash } from 'node:crypto'

/**
 * 团队项目映射。**解决一个必然会踩的坑**：
 *  本仓的 wsKey 是**按本机绝对路径派生**的（如 --D--dsh-auto-memory--）。
 *  两个人在不同盘符克隆同一个仓库 ⇒ 得到两个不同的 wsKey
 *  ⇒ 团队项目库**永远汇合不了**（各自写各自的分区）。
 *
 * 解法：引入与路径无关的 **teamProjectId**：
 *   优先级 ① 管理员显式登记（配置 teamProjectId）
 *          ② git remote.origin.url 的规范化哈希（同一个仓库 ⇒ 同一个 id）
 *          ③ fallback：本机 wsKey（退化为"仅本机"语义，并**明确告知用户**）
 */
export function createTeamProjectMap({ engine, execGitRemote, diag }) {
  let cached = null

  function derive() {
    // ① 显式登记优先
    const explicit = String(engine.config.teamProjectId || '').trim()
    if (explicit) return { id: explicit, src: 'explicit', stable: true }

    // ② git remote 派生（可选依赖：拿不到就跳）
    try {
      const url = execGitRemote && execGitRemote()      // 调方注入，插件不自己 spawn
      if (url && typeof url === 'string') {
        // 规范化：去协议差异（https/ssh 等价）、去 .git 后缀、去前后空白
        const norm = url.trim().toLowerCase()
          .replace(/^git@([^:]+):/, 'https://$1/')
          .replace(/^ssh:\/\//, 'https://')
          .replace(/\.git$/, '')
          .replace(/\/+$/, '')
        const id = 'gp_' + createHash('sha1').update(norm).digest('hex').slice(0, 16)
        return { id, src: 'git-remote', stable: true, url: norm }
      }
    } catch (e) { try { diag('project map git derive failed: ' + ((e && e.message) || e)) } catch (_) {} }

    // ③ fallback：本机 wsKey，**明确标为不稳定**
    const local = engine.currentWsKeyPre ? engine.currentWsKeyPre() : ''
    return { id: 'local_' + (local || 'unknown'), src: 'local-fallback', stable: false }
  }

  return {
    current() { if (!cached) cached = derive(); return cached },
    invalidate() { cached = null },
    /** 前端要显示的东西：来源 + 是否稳定（不稳定时必须提示用户） */
    describe() {
      const c = this.current()
      return {
        projectId: c.id,
        source: c.src,
        stable: c.stable,
        hint: c.stable ? '' : '未配置团队项目 ID 且无法从 git remote 派生：本项目的团队共享仅限本机。请在团队设置中登记项目 ID。',
      }
    },
  }
}
```

## 四、数据流：一次团队写入的完整旅程（端到端）

```
① 模型调用 memory_note（kind=plan/handoff，或 kind=note 且 scope=team）
   ↓
② team-envelope.wrap() 捕获                    [L4]
   ├─ identity.currentActor() → {id,name,role,device,ws}
   ├─ 先执行本体（既有写盘逻辑，零改动）        ← ★本地优先
   └─ scopeOf.isTeamWritable() 判定 → outbox.enqueue()
   ↓
③ outbox：幂等键去重 → lamport+1 → 同步落盘    [L3]
   ↓
④ team-sync.tick（挂在 pre-step 的 15s 节拍）  [L3]
   └─ queueMicrotask → outbox.flush() → fetchJson POST /v1/apply
        · 带 Authorization: Bearer <token>（team-auth 提供）
        · 带 x-dam-team: <teamId>           ← 租户隔离
        · 带 x-dam-client: <version>
   ↓
⑤ 服务端：按 teamId + projectId 归并，产出 CDC 游标
   ↓
⑥ 另一端（B 机器）pullOnce（同一 15s 节拍）    [L3]
   └─ applyOne(ch) → **走既有 upsert / scoped.write**   ← ★不绕过 Gate
        ├─ fact     → fact-store.upsert（六态：created/merged/superseded/conflict-added/...）
        ├─ procedure→ scoped 双库分派（物理分文件）
        └─ handoff  → 追加新篇（P2 语义，不覆盖）
   ↓
⑦ 冲突：三元组冲突 → conflictCenter.push（自动"保留双方"）
        整篇级冲突 → pushManual（人工裁决队列）
        被既有校验拒 → recordRejected（留痕，不静默）
   ↓
⑧ 注入：下一次 pre-step → renderTeamSnapshotPre 带版本号
        （版本未变返回空串 ⇒ 不击穿 DeepSeek 前缀缓存）
```
