## 依赖注入点（唯一需要改既有文件的地方）

在 `lib/index.js` 的 apply 阶段末尾（建议放在路由注册之后、`return` 之前）加一段**装配代码**：

```js
// ================= dam-team:begin =================
// 团队子系统装配。★ 全部在 teamEnabled 时才创建；关闭时**零对象、零文件、零网络**。
try {
  if (engine.config && engine.config.teamEnabled) {
    engine._teamIdentity = createTeamIdentity({ ctx, engine, diag })
    engine._teamIdentity.install()
    engine._teamProjectMap = createTeamProjectMap({
      engine, diag,
      execGitRemote: () => {                       // 注入 git 读取（插件不自己 spawn）
        try { return execFileSync('git', ['config', '--get', 'remote.origin.url'], { encoding: 'utf8' }) } catch (_) { return '' }
      },
    })
    engine._teamAuth = createTeamAuth({ ctx, engine, diag })
    engine._teamOutbox = createTeamOutbox({ engine, dir: engine.paths.memoryDir, diag })
    engine._teamOutbox.load()                       // 恢复上次未推完的
    engine._teamConflict = createConflictCenter({ engine, dir: engine.paths.memoryDir, diag })
    engine._teamPuller = createTeamPuller({ engine, fetchJson: engine._teamAuth.fetchJson, conflictCenter: engine._teamConflict, outbox: engine._teamOutbox, diag })
    engine._teamSync = createTeamSync({ engine, outbox: engine._teamOutbox, puller: engine._teamPuller, conflictCenter: engine._teamConflict, diag })
    engine._teamEnvelope = createTeamEnvelope({ engine, outbox: engine._teamOutbox, identity: engine._teamIdentity, scopeOf: teamScopeOf, diag })
    engine._teamState = createTeamState({ engine })   // state 路由用
    diag('team subsystem ready')
  }
} catch (eTeam) {
  // ★ 团队初始化失败**绝不能影响本机功能**
  console.error('[dsh-auto-memory] team init failed', eTeam && (eTeam.stack || eTeam.message) || eTeam)
  engine.config.teamEnabled = false                 // 自动降级，前端据此隐藏全部团队 UI
}
// ================= dam-team:end =================
```

**★ 关键判据**：`catch` 里**必须把 `teamEnabled` 置回 false** —— 让"团队初始化失败"自动降级为"单人模式"，而不是让插件半死。

## 装配辅助：`teamScopeOf`（判断哪次工具调用该同步）

```js
/**
 * 作用域裁决：决定一次工具调用是否产生团队变更。
 * ★ 默认**不**同步（保守）—— 只有明确列出的才上行。
 */
const TEAM_WRITABLE = Object.freeze({
  memory_note: (args) => {                          // 只有 plan/handoff/team 相关的才上行
    const k = args && args.kind
    return k === 'plan' || k === 'handoff' || (args && args.scope === 'team')
  },
  memory_user: () => true,                          // 用户级规则 = 团队治理面
  memory_rules: () => true,                         // 规则条目 = 团队治理面
  memory_procedure: () => true,                     // 技能 = 团队共享面
  calendar_add: (args) => (args && args.track === 'team'),
  calendar_done: (args) => (args && args.track === 'team'),
  calendar_remove: (args) => (args && args.track === 'team'),
})

const teamScopeOf = {
  isTeamWritable(toolName, args) {
    const f = TEAM_WRITABLE[toolName]
    if (!f) return false                            // ★ 未登记的一律不同步
    try { return !!f(args) } catch (_) { return false }
  },
}
```

**★ 为什么用白名单而非黑名单**：新增工具时**默认不同步**（保守），避免"新工具忘了归类 → 静默外传隐私"。

## 脱敏函数（出站前必做）

```js
/**
 * 出站脱敏。★ **必须在本机做**，不能指望服务端 —— 服务端不该看到本机路径。
 */
function redactForTeam(args) {
  const out = {}
  for (const [k, v] of Object.entries(args || {})) {
    if (typeof v === 'string') {
      // 去 Windows/Unix 绝对路径（含盘符与用户目录）
      let s = v.replace(/[A-Za-z]:\\[^\s"']+/g, '<path>')
               .replace(/\/(?:home|Users)\/[^\s"']+/g, '<path>')
      // 去可能的内网地址、token 形态串
      s = s.replace(/\b(?:10|172|192)\.\d+\.\d+\.\d+(?::\d+)?\b/g, '<host>')
           .replace(/\b(?:sk|ghp|gho|xox[baprs])-[A-Za-z0-9_-]{16,}\b/g, '<redacted>')
      out[k] = s
    } else if (Array.isArray(v)) {
      out[k] = v.map((x) => (typeof x === 'string' ? redactForTeam({ s: x }).s : x))
    } else if (v && typeof v === 'object') {
      out[k] = redactForTeam(v)
    } else {
      out[k] = v
    }
  }
  return out
}
```

**验收判据**：喂入含 `D:\dsh-auto-memory\lib\index.js`、`192.168.1.5:8080`、`sk-abcdefghijklmnop1234` 的样本 ⇒ 输出中三者均被替换。

## combine 进正式版的推荐步骤

```bash
# 1) 先做 B0 基线（冻结哈希 + 记录 180 全绿）
certutil -hashfile lib/index.js SHA256

# 2) 逐批落地（每批一提交，便于回退）
#    批 B1 → B2 → ... → B10（见 06 卷）

# 3) 每批后必跑
node --check lib/index.js
node tests/smoke/smoke-test.mjs      # 期望 180 PASS / 0 FAIL

# 4) ★ 每批后做「关闭态零差异」验证
#    把 teamEnabled 设回 false，注入文本应与 B0 逐字节一致
```

**★ 最重要的一条**：**不要一次性合并全部 9 个文件**。按 06 卷的批次走，每批都是一个可回退的独立状态。
理由：本仓有 **既有 smoke 回归（仓内自述 **180 套**；实测 `tests/smoke/` 下 **181 个 .mjs**，差额应为 1 个非套件辅助文件）**，一次性大改会让红灯无法归因（"哪一批引入的？"）。
